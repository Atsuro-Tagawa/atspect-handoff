# 12_ライブのモジュール（本番に出ている実物・pullで取得）

```js
/* ATSPECT 検索の正規化と照合 ── 2つの検索窓（作家を探す／作品を探す）で共有する唯一の実装。
 *
 * ★2026-08-09（SNS運用T・代表指示）＝それまで正規化が2ファイルに別々に書かれており、
 *   片方だけ直すともう片方に同じ穴が残る形だった。実際、次の食い違いが実測されている：
 *     「suzuki chikako」→ 作家を探す＝当たる／作品を探す＝落ちる
 *     「chikako suzuki」→ 作家を探す＝落ちる／作品を探す＝当たる
 *   （作家データのローマ字は「姓 名」＝Suzuki Chikako、作品データのローマ字は「名 姓」＝Chikako Suzuki のため）
 *   本ファイルに1本化し、両窓が同じ規則で照合する。
 *
 * 規則（確定設計）：
 *   ① 小文字化
 *   ② 空白（半角・全角）／ハイフン各種／中点（・ ･ ‧）を、語の区切りとして扱ってから除去
 *      ★長音符「ー」(U+30FC) はハイフンではないので残す（「ラーメン」を壊さないため）
 *   ③ マクロン吸収（ā ī ū ē ō → a i u e o）
 *   ④ カタカナ → ひらがな
 *   ⑤ 照合は語単位＝入力を区切りで分け、すべての語がどれかのキーに含まれれば一致
 *      → 姓だけ・名だけ・順序逆でも当たる／タイトルの語のどれでも当たる
 *   ⑥ キーにマクロンがある場合は「ō→o」の形に加えて「ō→ou」の形も別キーとして持つ
 *      → Funasōgaku / Funasogaku / Funasougaku のどれで打っても当たる
 *
 * 解かないもの（意図して未対応・注記で案内する）：
 *   ・漢字とかなの交ぜ書き（例＝「鈴木ちかこ」）＝読みがな（ひらがな）で打てば当たる
 *   ・簡体字（例＝「铃木」）＝当面対象外
 */
(function (w) {
  'use strict';
  if (w.ATSPECT_SEARCH) return;

  // 「丸ごと一致のときだけ当たるキー」の印（実データに現れない制御文字を使う）
  var EXACT = '/';
  var MACRON = { 'ā': 'a', 'ī': 'i', 'ū': 'u', 'ē': 'e', 'ō': 'o', 'â': 'a', 'î': 'i', 'û': 'u', 'ê': 'e', 'ô': 'o' };
  var MACRON_LONG = { 'ā': 'aa', 'ī': 'ii', 'ū': 'uu', 'ē': 'ei', 'ō': 'ou', 'â': 'aa', 'î': 'ii', 'û': 'uu', 'ê': 'ei', 'ô': 'ou' };
  // 区切りとみなす文字＝空白（半角/全角）・ハイフン各種・中点・アンダースコア・スラッシュ。
  // ★長音符「ー」(U+30FC) は含めない（「ラーメン」のような語を壊さないため）。
  // ★文字はすべてコードポイントで書く＝文字を直接並べると「　-‐」が範囲指定と解釈されて壊れるため
  //   （2026-08-09、実際にこの書き方で正規表現が例外になることを事前検査で見つけた）。
  var SEP_RE = /[\s\u3000\u002D\u2010\u2011\u2012\u2013\u2014\u2015\u2212\u30FB\uFF65\u2027\u00B7\u005F\u002F\u0027\u2019\u02BB\u02BC\u0060\u00B4]+/g;

  /* ★2026-08-09（Codexの指摘を受けて追加）＝表記のゆれを吸収する。
     ・NFKC＝全角英数字→半角、半角カタカナ→全角カタカナ
     ・結合文字で書かれたアクセント（例＝o＋結合マクロン）を、見た目どおりの1文字として扱う
     ・ラテン文字のアクセント（é ü ç 等）を素の文字に畳む
     ★かなの濁点・半濁点（U+3099/U+309A）は**落とさない**。落とすと「すずき」と「すすき」が
       同じ語になってしまい、別の作家に当たるため（この2つは 0300–036F の範囲外なので安全）。 */
  function foldWidthAndAccents(s) {
    var t = String(s || '').normalize('NFKC');
    return t.normalize('NFD').replace(/[\u0300-\u036F]/g, '').normalize('NFC');
  }
  function foldMacron(s, table) {
    return String(s || '').replace(/[āīūēōâîûêô]/g, function (c) { return table[c] || c; });
  }
  function kataToHira(s) {
    return String(s || '').replace(/[ァ-ヶ]/g, function (c) { return String.fromCharCode(c.charCodeAt(0) - 0x60); });
  }
  // 表示文字列 → 照合キー（区切りは落として1本につなげる）
  function norm(s) {
    return kataToHira(foldWidthAndAccents(foldMacron(String(s || '').toLowerCase(), MACRON))).replace(SEP_RE, '');
  }
  // マクロンを「ou/uu」式で書いた入力にも当てるための、キーのもう1つの形
  function normLong(s) {
    /* ★マクロンの置き換え（ō→ou）を先に済ませてから、残ったアクセントを畳む。
       順序を逆にすると ō が o になってしまい「ou」の形が作れない。 */
    return kataToHira(foldWidthAndAccents(foldMacron(String(s || '').toLowerCase(), MACRON_LONG))).replace(SEP_RE, '');
  }
  // 1つの値から照合キーを作る
  //   ・マクロンがあるときは「ō→o」の形と「ō→ou」の形の2本
  //   ・2語の値（人名を想定）は語順を入れ替えた形も作る
  //     ★理由＝同じ作家のローマ字が、作家データでは「姓 名」（Suzuki Chikako）、
  //       作品データでは「名 姓」（Chikako Suzuki）で入っている。空白を入れずに
  //       「suzukichikako」と打つと、どちらか片方の窓でしか当たらなかった（2026-08-09に実測）。
  //       データは触らずに、照合の側で両方の語順を持つことで2つの窓の挙動を揃える。
  function keysFrom(s) {
    var a = norm(s); if (!a) return [];
    var out = [a];
    var b = normLong(s);
    if (b && b !== a) out.push(b);
    var parts = String(s || '').trim().split(SEP_RE).filter(function (x) { return x.length > 0; });
    if (parts.length === 2) {
      // ★語順を入れ替えた形は「丸ごと一致のときだけ当たるキー」として持つ（先頭に EXACT を付けて印にする）。
      //   部分一致まで許すと、つなぎ目に元々無かった文字の並びができて誤ヒットが増える
      //   （実測＝610名で "Ota" が18件→35件に膨らんだ。丸ごと一致に限れば増分0件）。
      var r = norm(parts[1] + ' ' + parts[0]);
      if (r && r !== a && out.indexOf(EXACT + r) === -1) out.push(EXACT + r);
      var rl = normLong(parts[1] + ' ' + parts[0]);
      if (rl && rl !== r && out.indexOf(EXACT + rl) === -1) out.push(EXACT + rl);
    }
    return out;
  }
  // 複数の値をまとめて照合キーの配列にする
  function keysFromAll(values) {
    var out = [], i, k, j;
    for (i = 0; i < values.length; i++) { k = keysFrom(values[i]); for (j = 0; j < k.length; j++) if (out.indexOf(k[j]) === -1) out.push(k[j]); }
    return out;
  }
  // 入力 → 語の配列（区切りで分けてから各語を正規化）
  function tokens(q) {
    var raw = String(q || '').trim();
    if (!raw) return [];
    return raw.split(SEP_RE).map(function (t) { return norm(t); }).filter(function (t) { return t.length > 0; });
  }
  // 照合＝すべての語が、どれかのキーに当たること（順序は問わない）
  //   ふつうのキー＝部分一致／EXACT 印のキー＝丸ごと一致のときだけ当たる
  function matches(toks, keys) {
    if (!toks.length) return true;
    for (var i = 0; i < toks.length; i++) {
      var hit = false;
      for (var j = 0; j < keys.length; j++) {
        var k = keys[j];
        if (k.charAt(0) === EXACT) { if (k.slice(1) === toks[i]) { hit = true; break; } }
        else if (k.indexOf(toks[i]) !== -1) { hit = true; break; }
      }
      if (!hit) return false;
    }
    return true;
  }

  w.ATSPECT_SEARCH = {
    norm: norm, normLong: normLong, keysFrom: keysFrom, keysFromAll: keysFromAll,
    tokens: tokens, matches: matches, kataToHira: kataToHira
  };
})(window);
```
